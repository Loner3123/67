CREATE DATABASE pdm;
USE pdm;

-- Таблица пользователей
CREATE TABLE users (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL UNIQUE,
    password VARCHAR(50) NOT NULL,
    fullname VARCHAR(100),
    age INT,
    gender VARCHAR(20)
);

-- Таблица сообщений
CREATE TABLE usermessage (
    id INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(50) NOT NULL, -- Отправитель
    messege TEXT NOT NULL,
    userid INT, -- Для совместимости с твоим кодом (можно хранить id отправителя)
    touser VARCHAR(50) NOT NULL, -- Получатель
    date_sent DATETIME DEFAULT CURRENT_TIMESTAMP
);
